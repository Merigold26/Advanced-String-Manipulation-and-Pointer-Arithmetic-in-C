{

printf("\n%d",x);// x in değerini yazar

// pointerları yazdımak için %p kullanırız

// pointer dizinin ilk elemanına point eder yani sıfırıncı elemanına

ptr2=&ptr1;

printf("%d\n",*ptr1);
printf("%d\n",**ptr2);
printf("%p\n",ptr1);
printf("%p\n",ptr2);
printf("%p\n",&ptr2);

int a = 6;
int *ptr3 = &a;

printf("%p\n",ptr3);
